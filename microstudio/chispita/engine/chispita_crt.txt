// Soft CRT look: blurs the hard edges of pixels a little, adds a glow
// around bright colors and darkens the corners. No scanlines.
// Settings are in chispita_config: chispita_crt, chispita_crt_soften,
// chispita_crt_glow and chispita_crt_vignette.
//
// microScript can't read the screen's pixels, so the work is done in
// JavaScript on microStudio's screen canvas. It needs the default
// microStudio graphics (not Pixi, M2D, M3D or Babylon).

chispita_crt_ready = 0
chispita_crt_failed = 0

chispita_crt_setup = function()
  chispita_crt_ready = 1
  system.javascript('
  window.chispitaCrt = function(g) {
    var p = window.player;
    if (!p || !p.runtime || !p.runtime.screen) return 0;
    var src = p.runtime.screen.canvas;
    var ctx = p.runtime.screen.context;
    var W = src.width, H = src.height;
    if (!ctx || W < 2 || H < 2) return 0;
    var st = window.chispitaCrtState;
    if (!st) {
      st = window.chispitaCrtState = { bufs: {} };
    }
    var buf = function(name, w, h, quality) {
      w = Math.max(1, Math.round(w)); h = Math.max(1, Math.round(h));
      var b = st.bufs[name];
      if (!b) {
        b = document.createElement("canvas");
        b.c = b.getContext("2d");
        st.bufs[name] = b;
      }
      if (b.width != w || b.height != h) { b.width = w; b.height = h; }
      b.c.globalAlpha = 1;
      b.c.globalCompositeOperation = "copy";
      b.c.imageSmoothingEnabled = true;
      b.c.imageSmoothingQuality = quality;
      return b;
    };
    var num = function(v, d) { return (typeof v == "number" && isFinite(v)) ? v : d; };
    var unit = Math.min(W, H) / 200;
    var soften = Math.max(0, num(g.chispita_crt_soften, 0));
    var glow = Math.max(0, num(g.chispita_crt_glow, 0));
    var vignette = Math.max(0, num(g.chispita_crt_vignette, 0));
    if (soften <= 0 && glow <= 0 && vignette <= 0) return 1;
  
    // A smaller copy of the frame, drawn back stretched, blurs the hard pixel
    // edges. Its pixels are wider than tall, since a CRT smears sideways more.
    var soft = null, bloom = null;
    if (soften > 0 || glow > 0) {
      soft = buf("soft", W / unit, H / (unit * 0.5), "high");
      soft.c.drawImage(src, 0, 0, soft.width, soft.height);
    }
    if (glow > 0) {
      // Halve the soft copy step by step so the glow is smooth, not blocky.
      var from = soft, level = 0;
      while (from.height > 24 && W / from.width < unit * 6 && level < 8) {
        var b = buf("glow" + level, from.width / 2, from.height / 2, "low");
        b.c.drawImage(from, 0, 0, b.width, b.height);
        from = b; level++;
      }
      bloom = from;
      // Multiplying the glow by itself keeps bright colors and drops dark ones.
      bloom.c.globalCompositeOperation = "multiply";
      bloom.c.drawImage(bloom, 0, 0);
      bloom.c.drawImage(bloom, 0, 0);
    }
  
    ctx.save();
    ctx.setTransform(1, 0, 0, 1, 0, 0);
    ctx.imageSmoothingEnabled = true;
    ctx.globalCompositeOperation = "source-over";
    if (soft && soften > 0) {
      ctx.globalAlpha = Math.min(1, soften);
      ctx.drawImage(soft, 0, 0, W, H);
    }
    if (bloom) {
      // "screen" brightens like light does, without blowing colors out to white.
      ctx.globalCompositeOperation = "screen";
      ctx.globalAlpha = Math.min(1, glow);
      ctx.drawImage(bloom, 0, 0, bloom.width, bloom.height, 0, 0, W, H);
      ctx.globalCompositeOperation = "source-over";
    }
    if (vignette > 0) {
      if (!st.vig || st.vigW != W || st.vigH != H || st.vigA != vignette) {
        var r = Math.sqrt(W * W + H * H) / 2;
        var gr = ctx.createRadialGradient(W / 2, H / 2, r * 0.45, W / 2, H / 2, r);
        gr.addColorStop(0, "rgba(0,0,0,0)");
        gr.addColorStop(1, "rgba(0,0,0," + Math.min(1, vignette) + ")");
        st.vig = gr; st.vigW = W; st.vigH = H; st.vigA = vignette;
      }
      ctx.globalAlpha = 1;
      ctx.fillStyle = st.vig;
      ctx.fillRect(0, 0, W, H);
    }
    ctx.restore();
    return 1;
  };
  ')
end

Chispita_applyCrt = function()
  if chispita_crt_failed then return end
  if not chispita_crt_ready then chispita_crt_setup() end
  if not system.javascript("return window.chispitaCrt ? window.chispitaCrt(global) : 0") then
    chispita_crt_failed = 1
    chispita_warn("the CRT effect can't run here. It needs the default microStudio graphics, in a browser.")
  end
end
